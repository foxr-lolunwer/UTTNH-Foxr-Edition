FLTE_AI_build_land_facility = {
    enable = {
        land_facility < 1
        OR = {
			has_tech = FLTE_tech_tier_vehc_2
			has_tech = FLTE_tech_tier_ordn_2
		}
        num_of_civilian_factories > 20 # 保证有充足的民工用来建设
		num_of_military_factories > 10 # 没有军工建造这个没有意义
    }
    abort_when_not_enabled = yes
    ai_strategy = {
        type = build_building
        id = land_facility
        value = 100
    }
}

FLTE_AI_build_civilian_facility = {
    enable = {
        civilian_facility < 1
        num_of_civilian_factories > 5 # 能否建造建筑和该建筑有关，因此不能严格限制民工数量
        NOT = {
			any_controlled_state = {
				civilian_facility > 0
			}
		}
    }
    abort_when_not_enabled = yes
    ai_strategy = {
        type = build_building
        id = civilian_facility
        value = 1000 # 对AI发展至关重要，尤其是不能建造建筑的
    }
}

FLTE_AI_build_resource_expansion_steel = {
    enable = {
        has_tech = resource_expansion_steel_1
        num_of_civilian_factories > 30 # 国家需要有一定的工业才会考虑建造
        has_resources_in_country = {
            resource = steel
            amount < -20 # 低于-20代表该国家也买不到资源了（如果后面买到了，此策略会失效）。
        }
    }
    abort_when_not_enabled = yes
    ai_strategy = {
        type = build_building
        id = resource_expansion_steel
        value = 20
    }
}

FLTE_AI_build_industrial_expansion = { # 没有区分和平和战时倾向，因为战时也需要持续拓展建筑槽
    enable = {
        has_tech = capacity_industrial_construction
        OR = {
            num_of_civilian_factories > 20 # 保证有充足的民工用来建设
            num_of_controlled_states < 5 # 如果是微型国家，显然不能只判断民工数量
        }
    }
    abort_when_not_enabled = yes
    ai_strategy = {
        type = build_building
        id = industrial_expansion
        value = 5
    }
}
