FLTE_cheat_decision_categories = {
    FLTE_decision_get_infr_1_tech = {
        icon = generic_operation
        visible = {
            NOT = { OR = {
                has_tech = FLTE_tech_tier_infr_1
                has_tech = capacity_industrial_construction
            } }
            OR = {
                is_debug = yes
                has_global_flag = FLTE_GLFLAG_cheat_limited
                # has_global_flag = FLTE_GLFLAG_cheat_allowed allowed时同时获得limited
            }
        }
        complete_effect = {
            if = { limit = { NOT = { has_tech = FLTE_tech_tier_infr_1 } }
                set_technology = { FLTE_tech_tier_infr_1 = 1 }
            }
            if = { limit = { NOT = { has_tech = capacity_industrial_construction } }
                set_technology = { capacity_industrial_construction = 1 }
            }
        }
    }
